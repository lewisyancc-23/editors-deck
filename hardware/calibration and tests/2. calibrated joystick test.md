## This file is used to test the calibrated joystick.
Copy the code below and upload it to the Arduino using the Arduino IDE.

```text
const int JOY_X = A0;
const int JOY_Y = A1;

// Your calibrated center
const int CENTER_X = 506; #Change this to the value obtained from File 1. joystick calibration
const int CENTER_Y = 511; #Change this to the value obtained from File 1. joystick calibration

// Dead zone
const int DEADZONE = 150; #Change this to the value obtained from File 1. joystick calibration

void setup() {
  Serial.begin(115200);

  Serial.println("=== CALIBRATED JOYCON TEST ===");
  Serial.println("Move the joystick and check the detected direction.");
  Serial.println();
}

void loop() {
  int x = analogRead(JOY_X);
  int y = analogRead(JOY_Y);

  int dx = x - CENTER_X;
  int dy = y - CENTER_Y;

  // Inside dead zone
  if (abs(dx) < DEADZONE && abs(dy) < DEADZONE) {
    Serial.println("CENTER");
  }
  // Horizontal movement is dominant
  else if (abs(dx) > abs(dy)) {
    if (dx < -DEADZONE) {
      Serial.println("LEFT");
    }
    else if (dx > DEADZONE) {
      Serial.println("RIGHT");
    }
  }
  // Vertical movement is dominant
  else {
    if (dy < -DEADZONE) {
      Serial.println("UP");
    }
    else if (dy > DEADZONE) {
      Serial.println("DOWN");
    }
  }

  delay(200);
}
```
