## This file is used to calibrate the joystick. 
Copy the code below and upload it to the Arduino using the Arduino IDE.

```text
const int JOY_X = A0;
const int JOY_Y = A1;

void waitForMovement(const char* direction) {
  Serial.print("Move joystick ");
  Serial.print(direction);
  Serial.println(" and HOLD it...");

  delay(1500);

  int x = analogRead(JOY_X);
  int y = analogRead(JOY_Y);

  Serial.print(direction);
  Serial.print(" recorded: X = ");
  Serial.print(x);
  Serial.print(" | Y = ");
  Serial.println(y);

  delay(1000);
}

void setup() {
  Serial.begin(115200);

  Serial.println("================================");
  Serial.println("     JOYSTICK CALIBRATION");
  Serial.println("================================");
  Serial.println();

  Serial.println("Step 1: Leave joystick CENTERED.");
  Serial.println("Recording in 3 seconds...");

  delay(3000);

  int centerX = analogRead(JOY_X);
  int centerY = analogRead(JOY_Y);

  Serial.print("CENTER recorded: X = ");
  Serial.print(centerX);
  Serial.print(" | Y = ");
  Serial.println(centerY);

  Serial.println();
  delay(1500);

  waitForMovement("UP");
  waitForMovement("DOWN");
  waitForMovement("LEFT");
  waitForMovement("RIGHT");

  Serial.println();
  Serial.println("================================");
  Serial.println("       CALIBRATION DONE");
  Serial.println("================================");
}

void loop() {
}
```
