## This file is used to test the complete hardware. 
Copy the code below and upload it to the Arduino using the Arduino IDE.

```text
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

// =========================
// PIN DEFINITIONS
// =========================

const int JOY_X  = A0;
const int JOY_Y  = A1;
const int JOY_SW = 6;

const int BUTTON_1 = 2;
const int BUTTON_2 = 3;
const int BUTTON_3 = 4;
const int BUTTON_4 = 5;

// =========================
// JOYSTICK CALIBRATION
// =========================

const int CENTER_X = 506; #Change this to the value obtained from file 1. joystick calibration
const int CENTER_Y = 511; #Change this to the value obtained from file 1. joystick calibration

const int DEADZONE = 150; #Change this to the value obtained from file 1. joystick calibration

// =========================
// LCD
// =========================

// Most I2C 1602 modules use 0x27.
// If yours doesn't work, we'll scan the address.
LiquidCrystal_I2C lcd(0x27, 16, 2);

// =========================
// SETUP
// =========================

void setup() {

  Serial.begin(115200);

  // Joystick
  pinMode(JOY_SW, INPUT_PULLUP);

  // Buttons
  pinMode(BUTTON_1, INPUT_PULLUP);
  pinMode(BUTTON_2, INPUT_PULLUP);
  pinMode(BUTTON_3, INPUT_PULLUP);
  pinMode(BUTTON_4, INPUT_PULLUP);

  // LCD
  lcd.init();
  lcd.backlight();

  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("Editors Deck");

  lcd.setCursor(0, 1);
  lcd.print("Hardware Test");

  delay(2000);

  lcd.clear();

  Serial.println("================================");
  Serial.println("       EDITORS DECK TEST");
  Serial.println("================================");
}

// =========================
// JOYSTICK FUNCTION
// =========================

void checkJoystick() {

  int x = analogRead(JOY_X);
  int y = analogRead(JOY_Y);

  int dx = x - CENTER_X;
  int dy = y - CENTER_Y;

  // Joystick button
  if (digitalRead(JOY_SW) == LOW) {
    Serial.println("JOYCON BUTTON");
    lcd.setCursor(0, 1);
    lcd.print("JOY SW         ");
    delay(200);
    return;
  }

  // Center
  if (abs(dx) < DEADZONE && abs(dy) < DEADZONE) {
    return;
  }

  // Horizontal movement
  if (abs(dx) > abs(dy)) {

    if (dx < -DEADZONE) {
      Serial.println("LEFT");

      lcd.setCursor(0, 1);
      lcd.print("LEFT            ");
    }

    else if (dx > DEADZONE) {
      Serial.println("RIGHT");

      lcd.setCursor(0, 1);
      lcd.print("RIGHT           ");
    }
  }

  // Vertical movement
  else {

    if (dy < -DEADZONE) {
      Serial.println("UP");

      lcd.setCursor(0, 1);
      lcd.print("UP              ");
    }

    else if (dy > DEADZONE) {
      Serial.println("DOWN");

      lcd.setCursor(0, 1);
      lcd.print("DOWN            ");
    }
  }

  delay(150);
}

// =========================
// BUTTON FUNCTION
// =========================

void checkButtons() {

  if (digitalRead(BUTTON_1) == LOW) {

    Serial.println("BUTTON 1");

    lcd.setCursor(0, 1);
    lcd.print("BUTTON 1        ");

    delay(250);
  }

  if (digitalRead(BUTTON_2) == LOW) {

    Serial.println("BUTTON 2");

    lcd.setCursor(0, 1);
    lcd.print("BUTTON 2        ");

    delay(250);
  }

  if (digitalRead(BUTTON_3) == LOW) {

    Serial.println("BUTTON 3");

    lcd.setCursor(0, 1);
    lcd.print("BUTTON 3        ");

    delay(250);
  }

  if (digitalRead(BUTTON_4) == LOW) {

    Serial.println("BUTTON 4");

    lcd.setCursor(0, 1);
    lcd.print("BUTTON 4        ");

    delay(250);
  }
}

// =========================
// MAIN LOOP
// =========================

void loop() {

  checkJoystick();
  checkButtons();

}
```
